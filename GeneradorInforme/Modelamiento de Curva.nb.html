subamortiguada_T <- function(t_min,
                             T0 = 24,
                             Tfinal = -0.1,
                             PO = 0.10,
                             Ts = 90) {

  # 1) Calcular zeta desde overshoot
  f <- function(z) exp(-z*pi/sqrt(1 - z^2)) - PO
  zeta <- uniroot(f, c(0.01, 0.99))$root

  # 2) Frecuencia natural y amortiguada
  wn <- 4/(zeta*Ts)                 # rad/min
  wd <- wn*sqrt(1 - zeta^2)

  # 3) Respuesta subamortiguada
  alpha <- zeta/sqrt(1 - zeta^2)
  step01 <- 1 - exp(-zeta*wn*t_min) *
    (cos(wd*t_min) + alpha*sin(wd*t_min))

  # 4) Escalado a temperatura
  T <- T0 + (Tfinal - T0)*step01

  return(list(T = T, zeta = zeta, wn = wn, wd = wd))
}

curva_temp_v2 <- function(T0 = 24,
                          setpoint = -0.1,
                          duracion_proceso_min = 480,
                          pct_transitorio = 20,     # % del tiempo total
                          overshoot_pct = 10,       # % overshoot
                          osc_estable_pct = 2,      # % amplitud ondulaciones en estable
                          ruido_estable_pct = 0.6,  # % ruido (extra) en estable
                          n = 2000,
                          seed = 123) {

  set.seed(seed)

  # Tiempo
  t <- seq(0, duracion_proceso_min, length.out = n)

  # --- 1) Modelo 2° orden subamortiguado para el transitorio ---
  PO <- overshoot_pct / 100
  fz <- function(z) exp(-z*pi/sqrt(1 - z^2)) - PO
  zeta <- uniroot(fz, c(0.01, 0.99))$root

  t_trans_end <- (pct_transitorio / 100) * duracion_proceso_min
  if (t_trans_end <= 0) t_trans_end <- duracion_proceso_min * 0.1

  # Hacemos que el "settling" ocurra aprox al final del transitorio
  Ts_eff <- t_trans_end
  wn <- 4/(zeta * Ts_eff)
  wd <- wn * sqrt(1 - zeta^2)
  alpha <- zeta / sqrt(1 - zeta^2)

  step01 <- 1 - exp(-zeta * wn * t) * (cos(wd * t) + alpha * sin(wd * t))
  T_base <- T0 + (setpoint - T0) * step01

  # --- 2) “Estado estable” con ondulaciones + ruido (más notorio) ---
  # Escala de amplitud (si setpoint es muy pequeño, usa el salto)
  escala <- max(abs(setpoint), abs(setpoint - T0), 1)

  A_ond <- (osc_estable_pct / 100) * escala
  A_ruido <- (ruido_estable_pct / 100) * escala

  # Ondulaciones: mezcla de dos senos con fases aleatorias
  phi1 <- runif(1, 0, 2*pi)
  phi2 <- runif(1, 0, 2*pi)

  # Frecuencias (en rad/min) escogidas para “ruido ondulante” visible
  w1 <- 2*pi / 18   # ~ periodo 18 min
  w2 <- 2*pi / 7    # ~ periodo 7 min

  ond <- A_ond * (0.65*sin(w1*t + phi1) + 0.35*sin(w2*t + phi2))

  # Ruido aleatorio (gaussiano) + un poco de suavizado opcional
  ruido <- rnorm(length(t), mean = 0, sd = A_ruido)

  # Aplica ondulaciones/ruido SOLO después del fin del transitorio
  mask_estable <- t >= t_trans_end
  T <- T_base
  T[mask_estable] <- T_base[mask_estable] + ond[mask_estable] + ruido[mask_estable]

  # tp_objetivo (primer cruce con setpoint)
  tol <- 1e-6
  i_tp <- which(T <= setpoint + tol)[1]
  if (is.na(i_tp)) i_tp <- which.min(abs(T - setpoint))
  tp_obj <- t[i_tp]

  list(t = t,
       T = T,
       T_base = T_base,
       zeta = zeta, wn = wn, wd = wd,
       t_trans_end = t_trans_end,
       tp_obj = tp_obj,
       i_tp = i_tp,
       params = list(T0=T0, setpoint=setpoint, duracion_proceso_min=duracion_proceso_min,
                     pct_transitorio=pct_transitorio, overshoot_pct=overshoot_pct,
                     osc_estable_pct=osc_estable_pct, ruido_estable_pct=ruido_estable_pct,
                     n=n, seed=seed))
}

plot_transitorio_estable <- function(t, T, setpoint, t_corte,
                                     titulo = "Curva",
                                     banda_pct = 2) {

  # Banda ±2% alrededor del setpoint (manejo correcto si es negativo)
  band <- (banda_pct/100) * max(abs(setpoint), 1e-9)
  upper <- setpoint + band
  lower <- setpoint - band

  # Índices
  idx_trans <- t <= t_corte
  idx_est <- t >= t_corte

  op <- par(no.readonly = TRUE)
  par(mfrow = c(2,1), mar = c(4,4,3,1))

  # --- Transitorio ---
  plot(t[idx_trans], T[idx_trans], type="l", lwd=2, col="blue",
       xlab="Tiempo [min]", ylab="Temperatura [°C]",
       main=paste0(titulo, " — Estado transitorio (0 a ", round(t_corte,2), " min)"))
  abline(h=setpoint, col="red", lty=2)
  abline(v=t_corte, col="black", lty=2)
  grid()

  # --- Estable (zoom) ---
  t2 <- t[idx_est]
  T2 <- T[idx_est]

  yr <- range(T2, upper, lower)
  zoom_pad <- max((yr[2]-yr[1])*0.2, band*6, 1e-6)
  ylim2 <- c(yr[1]-zoom_pad, yr[2]+zoom_pad)

  plot(t2, T2, type="l", lwd=2, col="darkgreen",
       xlab="Tiempo [min]", ylab="Temperatura [°C]",
       main=paste0(titulo, " — Estado estable (zoom)"),
       ylim=ylim2)
  abline(h=setpoint, col="red", lty=2)
  abline(h=c(lower, upper), col="gray", lty=3)
  abline(v=t_corte, col="black", lty=2)
  grid()

  par(op)
}

detectar_regimen <- function(t, T,
                             banda_pct = 2,          # ±2% por defecto
                             tail_pct = 20,          # % final para estimar estado estable
                             ventana_min = 10,       # minutos consecutivos dentro de banda
                             dt_min = NULL) {        # si no lo das, se estima

  stopifnot(length(t) == length(T), length(t) >= 5)

  # Estimar dt (asumiendo muestreo casi uniforme)
  if (is.null(dt_min)) {
    dt_min <- median(diff(t))
  }

  # Cuántos puntos equivalen a ventana_min
  k <- max(1, round(ventana_min / dt_min))

  # 1) Estimar valor final con el último tail_pct%
  n <- length(T)
  tail_n <- max(5, floor((tail_pct / 100) * n))
  T_final_est <- mean(T[(n - tail_n + 1):n], na.rm = TRUE)

  # 2) Banda ±banda_pct alrededor del valor final
  #    Si el valor final es muy cercano a 0, usamos escala por rango para evitar banda cero
  escala <- max(abs(T_final_est), diff(range(T)), 1e-6)
  band <- (banda_pct / 100) * escala
  lower <- T_final_est - band
  upper <- T_final_est + band

  dentro <- (T >= lower) & (T <= upper)

  # 3) Primer índice donde se mantiene dentro por k puntos seguidos
  #    Usamos running sum de TRUEs
  dentro_num <- as.integer(dentro)
  runsum <- stats::filter(dentro_num, rep(1, k), sides = 1)

  i_est <- which(runsum >= k)[1]
  if (is.na(i_est)) {
    # No se encontró estado estable según criterio
    return(list(
      estable_encontrado = FALSE,
      t_estable_inicio = NA_real_,
      i_estable_inicio = NA_integer_,
      T_final_est = T_final_est,
      banda = c(lower = lower, upper = upper),
      k = k,
      dt_min = dt_min,
      transitorio = list(t = t, T = T),
      estable = list(t = numeric(0), T = numeric(0))
    ))
  }

  # Ajuste: i_est es el final de la primera ventana; el inicio real es:
  i_ini <- max(1, i_est - k + 1)

  list(
    estable_encontrado = TRUE,
    t_estable_inicio = t[i_ini],
    i_estable_inicio = i_ini,
    T_final_est = T_final_est,
    banda = c(lower = lower, upper = upper),
    k = k,
    dt_min = dt_min,
    transitorio = list(t = t[1:(i_ini - 1)], T = T[1:(i_ini - 1)]),
    estable = list(t = t[i_ini:n], T = T[i_ini:n])
  )
}
